# Q1a: A* Search Algorithm
import heapq

def a_star_search(graph, start, goal, heuristic):
    open_list = [(heuristic[start], 0, start, [start])]
    visited = set()
    
    while open_list:
        f, g, node, path = heapq.heappop(open_list)
        
        if node == goal:
            return path, g
        
        if node in visited:
            continue
        visited.add(node)
        
        for neighbor, cost in graph.get(node, []):
            if neighbor not in visited:
                new_g = g + cost
                new_f = new_g + heuristic.get(neighbor, 0)
                heapq.heappush(open_list, (new_f, new_g, neighbor, path + [neighbor]))
    
    return None, float('inf')

# Example graph
graph = {
    'A': [('B', 1), ('C', 4)],
    'B': [('D', 2), ('E', 5)],
    'C': [('E', 2)],
    'D': [('F', 3)],
    'E': [('F', 1)],
    'F': []
}

heuristic = {'A': 6, 'B': 4, 'C': 4, 'D': 3, 'E': 2, 'F': 0}

path, cost = a_star_search(graph, 'A', 'F', heuristic)
print(f"A* Path: {path}, Cost: {cost}")

def greedy_best_first_search(graph, start, goal, heuristic):
    open_list = [(heuristic[start], start, [start])]
    visited = set()
    
    while open_list:
        h, node, path = heapq.heappop(open_list)
        
        if node == goal:
            return path
        
        if node in visited:
            continue
        visited.add(node)
        
        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                heapq.heappush(open_list, (heuristic.get(neighbor, 0), neighbor, path + [neighbor]))
    
    return None

# Q3a: Simple Linear Regression using Least Squares Method
X = np.array([1, 2, 3, 4, 5])
Y = np.array([2, 4, 5, 4, 5])

X_mean = np.mean(X)
Y_mean = np.mean(Y)

numerator = np.sum((X - X_mean) * (Y - Y_mean))
denominator = np.sum((X - X_mean) ** 2)

m = numerator / denominator
b = Y_mean - m * X_mean

print(f"\nLinear Regression (Least Squares): Y = {m:.2f}X + {b:.2f}")

X_new = np.array([6, 7])
Y_pred = m * X_new + b
print(f"Predictions for X={X_new}: {Y_pred}")

# Q4a: Breadth First Search (BFS)
from collections import deque

def bfs(graph, start, goal):
    queue = deque([(start, [start])])
    visited = set()
    
    while queue:
        node, path = queue.popleft()
        
        if node == goal:
            return path
        
        if node in visited:
            continue
        visited.add(node)
        
        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                queue.append((neighbor, path + [neighbor]))
    
    return None

graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['E'],
    'D': ['F'],
    'E': ['F'],
    'F': []
}

path = bfs(graph, 'A', 'F')
print(f"\nBFS Path: {path}")

# Q4b: Prolog Facts (Python simulation)
facts = [
    ("likes", "ram", "music"),
    ("likes", "ram", "reading"),
    ("likes", "shyam", "music"),
    ("likes", "shyam", "gaming"),
    ("likes", "ajay", "sports"),
    ("student", "ram"),
    ("student", "shyam"),
    ("student", "ajay"),
]

def query(predicate, *args):
    results = []
    for fact in facts:
        if len(fact) == len((predicate,) + args):
            if fact[0] == predicate:
                match = True
                for i, arg in enumerate(args):
                    if arg != fact[i+1] and arg != '_':
                        match = False
                        break
                if match:
                    results.append(fact)
    return results

print("\nQ4b: Prolog-like Facts in Python")
print("\nFacts defined:")
for f in facts:
    print(f"  {f[0]}({f[1]}, {f[2]})" if len(f) > 2 else f"  {f[0]}({f[1]})")

print("\nQuery: likes(ram, X)")
print("Results:", query("likes", "ram", "_"))

print("\nQuery: likes(X, music)")
print("Results:", query("likes", "_", "music"))

# Q5a: Depth First Search (DFS)
def dfs(graph, start, goal, visited=None, path=None):
    if visited is None:
        visited = set()
    if path is None:
        path = []
    
    path.append(start)
    visited.add(start)
    
    if start == goal:
        return path
    
    for neighbor in graph.get(start, []):
        if neighbor not in visited:
            result = dfs(graph, neighbor, goal, visited, path)
            if result:
                return result
    
    path.pop()
    return None

graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['E'],
    'D': ['F'],
    'E': ['F'],
    'F': []
}

path = dfs(graph, 'A', 'F')
print(f"\nDFS Path: {path}")

# Q5b: Linear Regression using Gradient Descent Method
X = np.array([1, 2, 3, 4, 5])
Y = np.array([2, 4, 5, 4, 5])

m = 0
b = 0
learning_rate = 0.01
n = len(X)
iterations = 1000

for i in range(iterations):
    Y_pred = m * X + b
    dm = (-2/n) * sum(X * (Y - Y_pred))
    db = (-2/n) * sum(Y - Y_pred)
    m = m - learning_rate * dm
    b = b - learning_rate * db

print(f"\nLinear Regression (Gradient Descent): Y = {m:.4f}X + {b:.4f}")

X_new = np.array([6, 7])
Y_pred = m * X_new + b
print(f"Predictions for X={X_new}: {Y_pred}")

# Q6a: Family Tree (Prolog simulation in Python)
family = {
    'male': ['john', 'bob', 'tom', 'jim'],
    'female': ['alice', 'beth', 'susan', 'emma'],
    'parent': [
        ('bob', 'tom'), ('bob', 'emma'),
        ('alice', 'tom'), ('alice', 'emma'),
        ('john', 'jim'), ('beth', 'jim')
    ],
}

def father(child, parent):
    return parent in family['male'] and (parent, child) in family['parent']

def mother(child, parent):
    return parent in family['female'] and (parent, child) in family['parent']

def sibling(x, y):
    return x != y and any(
        p1 == p2 and c1 == x and c2 == y
        for (p1, c1) in family['parent']
        for (p2, c2) in family['parent']
    )

print("\nQ6a: Family Tree Queries")
print("\nMales:", family['male'])
print("Females:", family['female'])
print("\nParent-Child relations:", family['parent'])

print("\nIs bob father of tom?", father('tom', 'bob'))
print("Is alice mother of emma?", mother('emma', 'alice'))
print("Are tom and emma siblings?", sibling('tom', 'emma'))
